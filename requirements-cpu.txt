#   pip install torch torchvision --index-url https://download.pytorch.org/whl/cpu
#   pip install -r requirements-cpu.txt

# Install torch FIRST from that index URL. torch is deliberately not listed
# below: the default wheel bundles ~2.5 GB of CUDA libraries a CPU-only box
# cannot use.
#
# Keep config.yaml pointing at models/twinlitenet_int8.onnx here -- INT8
# dynamic quantization is a CPU optimisation and runs 5.4x faster than FP32.
#
# Do NOT also install onnxruntime-gpu. Both wheels ship the same
# onnxruntime/capi/*.so, so the second overwrites the first while pip still
# lists both, and you silently get whichever came last.
